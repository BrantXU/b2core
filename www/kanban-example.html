KanbanRender 看板渲染器

基于 TableRender 的看板式数据渲染组件，支持拖拽排序和浏览器缓存

使用代码示例

// 初始化看板
const kanban = new KanbanRender('myKanban', {
    data: [
        {
            id: '1',
            title: '设计用户界面',
            description: '完成登录页面的UI设计',
            status: '进行中',
            priority: '高',
            assignee: '张三',
            dueDate: '2024-01-15'
        },
        {
            id: '2', 
            title: '开发API接口',
            description: '实现用户管理相关的RESTful API',
            status: '待处理',
            priority: '中',
            assignee: '李四',
            dueDate: '2024-01-20'
        }
    ],
    categoryField: 'status',
    defaultCategories: ['待处理', '进行中', '已完成', '已暂停'],
    showSearch: true,
    showCreate: true,
    enableDrag: true,
    enableCache: true,
    cacheKey: 'my_kanban_state'
});

// 设置分类字段
kanban.setCategoryField('priority');

// 添加新分类
kanban.addCategory('紧急任务');

// 刷新数据
kanban.refresh();

// 获取看板数据
const kanbanData = kanban.getData();
